    <script>
        // Handle screen transitions
        function nextSlide(currentIndex) {
            // Play background music on first tap if you configured it
            if (currentIndex === 1) {
                const music = document.getElementById('bg-music');
                if (music) music.play().catch(e => console.log("Audio play blocked until interaction"));
                // Start continuous floating hearts
                setInterval(createHeart, 400);
            }

            // Move to next screen
            document.getElementById(`slide${currentIndex}`).classList.remove('active');
            const next = document.getElementById(`slide${currentIndex + 1}`);
            next.classList.add('active');

            // Trigger confetti if it's the final birthday screen
            if (currentIndex === 3) {
                triggerConfettiExplosion();
            }
        }

        // Generate floating background hearts
        function createHeart() {
            const container = document.getElementById('heart-container');
            const heart = document.createElement('div');
            heart.classList.add('heart-float');
            heart.innerText = ['❤️', '💖', '💝', '💕', '🌸'][Math.floor(Math.random() * 5)];
            
            // Randomize position and timing
            heart.style.left = Math.random() * 100 + 'vw';
            heart.style.animationDuration = (Math.random() * 2 + 3) + 's';
            heart.style.fontSize = (Math.random() * 20 + 15) + 'px';
            
            container.appendChild(heart);
            
            // Clean up heart after animation finish
            setTimeout(() => { heart.remove(); }, 5000);
        }

        // Confetti explosion physics
        function triggerConfettiExplosion() {
            var duration = 5 * 1000;
            var end = Date.now() + duration;

            (function frame() {
                confetti({
                    particleCount: 3,
                    angle: 60,
                    spread: 55,
                    origin: { x: 0, y: 0.8 }
                });
                confetti({
                    particleCount: 3,
                    angle: 120,
                    spread: 55,
                    origin: { x: 1, y: 0.8 }
                });

                if (Date.now() < end) {
                    requestAnimationFrame(frame);
                }
            }());
        }
    </script>
